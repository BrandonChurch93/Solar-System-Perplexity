import * as THREE from 'three'
import type { LensLook } from './lens'

export interface GhostSpec { name: string; at: number; size: number; long: number; soft: number; ring: number; fringe: number; round: number; color: number[] }
export interface GhostFit { sun: number[]; blades: number; turnDeg: number; round: number; edgePx: number; ghosts: GhostSpec[] }

const MAX = 12

const VERT = /* glsl */ `
out vec2 vNdc;
void main() {
  vNdc = position.xy;
  gl_Position = vec4(position.xy, 0.0, 1.0);
}`

const FRAG = /* glsl */ `
precision highp float;
uniform vec2 uSun;
uniform float uAmount;
uniform float uGain;
uniform float uAspect;
uniform float uVignette;
uniform vec4 uBody;
uniform vec3 uBlades;
uniform int uCount;
uniform vec4 uPlace[${MAX}];
uniform vec4 uTint[${MAX}];
uniform vec2 uForm[${MAX}];
uniform int uDiag;
in vec2 vNdc;
out vec4 outColor;
float opening(vec2 q, float round) {
  float a = atan(q.y, q.x) + uBlades.y;
  float seg = 6.283185307 / uBlades.x;
  float side = cos(floor(0.5 + a / seg) * seg - a) / cos(seg * 0.5);
  return length(q) * mix(side, 1.0, round);
}
void main() {
  vec2 p = vNdc * vec2(uAspect, 1.0);
  float d = dot(uSun, uSun);
  vec2 axis = uSun / sqrt(max(d, 1e-8));
  float px = fwidth(vNdc.y) * uBlades.z;
  vec3 sum = vec3(0.0);
  for (int i = 0; i < ${MAX}; i++) {
    if (i >= uCount) break;
    vec4 g = uPlace[i];
    vec2 q = p - uSun * g.x;
    q -= axis * (dot(q, axis) * (1.0 - 1.0 / g.z));
    float r = opening(q, uForm[i].y) / g.y;
    float w = max(g.w, px / g.y);
    vec3 rr = r / (1.0 + uForm[i].x * vec3(1.0, 0.0, -1.0));
    sum += uTint[i].rgb * (1.0 - smoothstep(1.0 - w, 1.0 + w, rr)) * (1.0 + uTint[i].a * smoothstep(0.7, 1.0, r));
  }
  vec2 v = vNdc * 0.5 * vec2(uAspect, 1.0);
  float onBody = 1.0 - smoothstep(uBody.z * 0.93, uBody.z, length(p - uBody.xy));
  float gain = uGain * uAmount * (1.0 - uBody.w * onBody) / mix(1.0, 1.0 / pow(1.0 + dot(v, v) * 0.9, 2.0), uVignette);
  outColor = vec4(sum * gain, 0.0);
  if (uDiag >= 2 || uAmount <= 0.0 || d < 1e-4) discard;
}`

export class Ghosts {
  readonly mesh: THREE.Mesh
  readonly material: THREE.ShaderMaterial
  private readonly v = new THREE.Vector3()
  private readonly q = new THREE.Quaternion()

  constructor(fit: GhostFit, private readonly look: LensLook, diag: THREE.IUniform<number>) {
    const list = fit.ghosts.slice(0, MAX)
    const pad = <T>(make: (g: GhostSpec | undefined) => T): T[] => Array.from({ length: MAX }, (_, i) => make(list[i]))
    this.material = new THREE.ShaderMaterial({
      glslVersion: THREE.GLSL3, vertexShader: VERT, fragmentShader: FRAG, depthTest: false, depthWrite: false, transparent: true,
      blending: THREE.CustomBlending, blendEquation: THREE.AddEquation, blendSrc: THREE.OneFactor, blendDst: THREE.OneFactor, blendSrcAlpha: THREE.ZeroFactor, blendDstAlpha: THREE.OneFactor,
      uniforms: {
        uSun: { value: new THREE.Vector2() }, uAmount: { value: 0 }, uGain: { value: 1 }, uAspect: { value: 16 / 9 }, uVignette: { value: 0 }, uDiag: diag,
        uBody: { value: new THREE.Vector4(0, 0, 1, 0) }, uBlades: { value: new THREE.Vector3(fit.blades, THREE.MathUtils.degToRad(fit.turnDeg), fit.edgePx) }, uCount: { value: list.length },
        uPlace: { value: pad((g) => new THREE.Vector4(g?.at ?? 0, g?.size ?? 1, g?.long ?? 1, g?.soft ?? 1)) },
        uTint: { value: pad((g) => new THREE.Vector4(g?.color[0] ?? 0, g?.color[1] ?? 0, g?.color[2] ?? 0, g?.ring ?? 0)) },
        uForm: { value: pad((g) => new THREE.Vector2(g?.fringe ?? 0, g?.round ?? fit.round)) },
      },
    })
    this.mesh = new THREE.Mesh(new THREE.PlaneGeometry(2, 2), this.material)
    this.mesh.frustumCulled = false
  }

  update(sun: THREE.Vector3, camera: THREE.PerspectiveCamera, amount: number, gain: number, vignette: number, aspect: number): void {
    const u = this.material.uniforms
    const v = this.v.copy(sun).applyQuaternion(this.q.copy(camera.quaternion).invert())
    const tan = Math.tan(THREE.MathUtils.degToRad(camera.fov) / 2)
    let seen = 0
    if (v.z < -1e-3) {
      const x = v.x / -v.z / tan
      const y = v.y / -v.z / tan
      ;(u.uSun!.value as THREE.Vector2).set(x, y)
      const out = Math.max(Math.abs(x) / aspect, Math.abs(y))
      seen = 1 - THREE.MathUtils.smoothstep(out, this.look.fade[0], this.look.fade[1])
    }
    u.uAmount!.value = amount * seen * this.look.gain
    u.uGain!.value = gain
    u.uVignette!.value = vignette
    u.uAspect!.value = aspect
    this.mesh.visible = amount * seen > 0
  }

  shade(body: THREE.Vector3, camera: THREE.PerspectiveCamera, keep: number): void {
    const u = this.material.uniforms.uBody!.value as THREE.Vector4
    const v = this.v.copy(body).applyQuaternion(this.q.copy(camera.quaternion).invert())
    const d = v.length()
    if (v.z > -1e-3 || d <= 1 || keep >= 1) {
      u.set(0, 0, 1, 0)
      return
    }
    const tan = Math.tan(THREE.MathUtils.degToRad(camera.fov) / 2)
    const off = Math.hypot(v.x, v.y)
    const theta = Math.atan2(off, -v.z)
    const alpha = Math.asin(1 / d)
    const hi = Math.tan(Math.min(theta + alpha, 1.5))
    const lo = Math.tan(theta - alpha)
    const k = off > 1e-9 ? (hi + lo) / 2 / off : 0
    u.set((v.x * k) / tan, (v.y * k) / tan, (hi - lo) / 2 / tan, 1 - keep)
  }

  dispose(): void {
    this.mesh.geometry.dispose()
    this.material.dispose()
  }
}
